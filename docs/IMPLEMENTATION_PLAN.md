```python
"""Idempotent DB bootstrap: create advdatafinal database + schemas + pgvector."""
import os
from dotenv import load_dotenv
import psycopg2

load_dotenv()

PG = dict(
    host=os.getenv("PG_HOST"),
    port=os.getenv("PG_PORT"),
    user=os.getenv("PG_USER"),
    password=os.getenv("PG_PASSWORD"),
)

def create_db():
    conn = psycopg2.connect(database="postgres", **PG)
    conn.autocommit = True
    cur = conn.cursor()
    cur.execute("SELECT 1 FROM pg_database WHERE datname = 'advdatafinal'")
    if not cur.fetchone():
        cur.execute("CREATE DATABASE advdatafinal")
        print("Created database advdatafinal")
    else:
        print("Database advdatafinal already exists")
    conn.close()

def init_schemas():
    conn = psycopg2.connect(database="advdatafinal", **PG)
    conn.autocommit = True
    cur = conn.cursor()
    for schema in ("raw", "datos_masked", "silver", "gold"):
        cur.execute(f"CREATE SCHEMA IF NOT EXISTS {schema}")
    cur.execute("CREATE EXTENSION IF NOT EXISTS vector")
    cur.execute("""
        CREATE TABLE IF NOT EXISTS raw.ingest_log (
          ingest_id    bigserial PRIMARY KEY,
          source       text NOT NULL,
          symbol       text,
          window_key   text NOT NULL,
          file_path    text NOT NULL,
          row_count    integer,
          sha256       char(64) NOT NULL,
          ingested_at  timestamptz NOT NULL DEFAULT now(),
          UNIQUE (source, symbol, window_key)
        )
    """)
    conn.close()
    print("Schemas + pgvector + ingest_log ready")

if __name__ == "__main__":
    create_db()
    init_schemas()
```

```python
"""Shared helpers: env loading, Postgres connection, ingest_log writer, sha256."""
from __future__ import annotations

import hashlib
import os
from pathlib import Path
from typing import Any

import psycopg2
from dotenv import load_dotenv

load_dotenv()

BRONZE_ROOT = Path(__file__).parent.parent / "bronze"
BRONZE_ROOT.mkdir(exist_ok=True)


def pg_conn(dbname: str = "advdatafinal") -> psycopg2.extensions.connection:
    return psycopg2.connect(
        host=os.getenv("PG_HOST"),
        port=os.getenv("PG_PORT"),
        user=os.getenv("PG_USER"),
        password=os.getenv("PG_PASSWORD"),
        database=dbname,
    )


def sha256_of_file(path: Path) -> str:
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(8192), b""):
            h.update(chunk)
    return h.hexdigest()


def log_ingest(
    source: str,
    symbol: str | None,
    window_key: str,
    file_path: Path,
    row_count: int,
) -> None:
    """Idempotent upsert into raw.ingest_log."""
    sha = sha256_of_file(file_path)
    with pg_conn() as conn, conn.cursor() as cur:
        cur.execute(
            """
            INSERT INTO raw.ingest_log
              (source, symbol, window_key, file_path, row_count, sha256, ingested_at)
            VALUES (%s, %s, %s, %s, %s, %s, now())
            ON CONFLICT (source, symbol, window_key) DO UPDATE SET
              file_path = EXCLUDED.file_path,
              row_count = EXCLUDED.row_count,
              sha256    = EXCLUDED.sha256,
              ingested_at = now();
            """,
            (source, symbol, window_key, str(file_path), row_count, sha),
        )


def load_universe() -> dict[str, list[str]]:
    """Return {sector_name: [ticker, ...]} from config/universe.yaml."""
    import yaml

    cfg = yaml.safe_load(open(Path(__file__).parent.parent / "config" / "universe.yaml"))
    return cfg["universe"]["tickers"]


def all_tickers() -> list[str]:
    return [t for tickers in load_universe().values() for t in tickers]
```

```python
"""Pull daily OHLCV for the 20-ticker universe via yfinance."""
from __future__ import annotations

import pandas as pd
import yfinance as yf
from tqdm import tqdm

from ingest.common import BRONZE_ROOT, all_tickers, log_ingest, pg_conn


def fetch_one(symbol: str, start: str = "2021-01-01", end: str = "2026-01-01") -> pd.DataFrame:
    df = yf.download(symbol, start=start, end=end, interval="1d", progress=False, auto_adjust=False)
    if isinstance(df.columns, pd.MultiIndex):
        df.columns = df.columns.get_level_values(0)
    df = df.reset_index()
    df.columns = [c.lower().replace(" ", "_") for c in df.columns]
    df["symbol"] = symbol
    return df


def land_to_bronze(symbol: str, df: pd.DataFrame) -> None:
    out = BRONZE_ROOT / "prices" / f"{symbol}.parquet"
    out.parent.mkdir(parents=True, exist_ok=True)
    df.to_parquet(out, index=False)
    log_ingest("yfinance_prices", symbol, "2021-2025", out, len(df))


def land_to_raw(symbol: str, df: pd.DataFrame) -> None:
    with pg_conn() as conn, conn.cursor() as cur:
        cur.execute(
            """
            CREATE TABLE IF NOT EXISTS raw.prices_raw (
              symbol     text,
              trade_date text,
              open       text,
              high       text,
              low        text,
              close      text,
              adj_close  text,
              volume     text,
              ingest_ts  timestamptz NOT NULL DEFAULT now(),
              PRIMARY KEY (symbol, trade_date)
            )
            """
        )
        rows = [
            (
                symbol,
                r["date"].strftime("%Y-%m-%d"),
                str(r["open"]),
                str(r["high"]),
                str(r["low"]),
                str(r["close"]),
                str(r["adj_close"]),
                str(int(r["volume"])),
            )
            for _, r in df.iterrows()
        ]
        cur.executemany(
            """
            INSERT INTO raw.prices_raw (symbol, trade_date, open, high, low, close, adj_close, volume)
            VALUES (%s, %s, %s, %s, %s, %s, %s, %s)
            ON CONFLICT (symbol, trade_date) DO UPDATE SET
              open      = EXCLUDED.open,
              high      = EXCLUDED.high,
              low       = EXCLUDED.low,
              close     = EXCLUDED.close,
              adj_close = EXCLUDED.adj_close,
              volume    = EXCLUDED.volume,
              ingest_ts = now()
            """,
            rows,
        )


def main() -> None:
    for symbol in tqdm(all_tickers(), desc="prices"):
        df = fetch_one(symbol)
        land_to_bronze(symbol, df)
        land_to_raw(symbol, df)


if __name__ == "__main__":
    main()
```

```python
def land_one(symbol: str, endpoint: str, payload: list[dict]) -> None:

    out = BRONZE_ROOT / "fund" / endpoint / f"{symbol}.json"
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(json.dumps(payload, indent=2))

    if not payload:
        log_ingest(f"fmp_{endpoint}", symbol, "2021-2025", out, 0)
        return

    # Create the raw table dynamically based on first payload row keys
    keys = list(payload[0].keys())
    cols_def = ", ".join(f'"{k}" text' for k in keys)
    pk = '"symbol", "date"' if "symbol" in keys and "date" in keys else '"symbol"'
    with pg_conn() as conn, conn.cursor() as cur:
        cur.execute(
            f"""
            CREATE TABLE IF NOT EXISTS {table} (
              {cols_def},
              ingest_ts timestamptz NOT NULL DEFAULT now(),
              PRIMARY KEY ({pk})
            )
            """
        )
        for row in payload:
            placeholders = ", ".join(["%s"] * len(keys))
            update_clause = ", ".join(f'"{k}" = EXCLUDED."{k}"' for k in keys if k not in {"symbol", "date"})
            cur.execute(
                f"""
                INSERT INTO {table} ({", ".join(f'"{k}"' for k in keys)})
                VALUES ({placeholders})
                ON CONFLICT ({pk}) DO UPDATE SET {update_clause}, ingest_ts = now()
                """,
                tuple(str(row.get(k)) if row.get(k) is not None else None for k in keys),
            )
    log_ingest(f"fmp_{endpoint}", symbol, "2021-2025", out, len(payload))


def main() -> None:
    for symbol in tqdm(all_tickers(), desc="fundamentals"):
        for endpoint in ENDPOINT_TABLE:
            payload = fetch_one(symbol, endpoint)
            land_one(symbol, endpoint, payload)
            sleep(0.25)  # gentle on FMP rate limits

    main()
```

```python
def fetch_10k_item1a(symbol: str, years: int = 5) -> list[dict]:
    c = Company(symbol)
    filings = c.get_filings(form="10-K").latest(years)
    out = []
    for f in filings:
        try:
            tenk = f.obj()  # parsed 10-K
            risk = tenk.items.get("Item 1A") or tenk.items.get("Item 1A.")
            out.append(
                dict(
                    accession=f.accession_no,
                    filing_date=str(f.filing_date),
                    fiscal_year=getattr(tenk, "period_of_report", None),
                    body=str(risk) if risk else "",
                )
            )
        except Exception as e:
            print(f"  10-K parse failed for {symbol} ({f.accession_no}): {e}")
    return out


def fetch_8k(symbol: str, months: int = 12) -> list[dict]:
    from datetime import date, timedelta
    cutoff = date.today() - timedelta(days=30 * months)
    c = Company(symbol)
    filings = c.get_filings(form="8-K")
    out = []
    for f in filings:
        if f.filing_date < cutoff:
            break
        try:
            out.append(
                dict(
                    accession=f.accession_no,
                    filing_date=str(f.filing_date),
                    body=str(f.obj()),
                )
            )
        except Exception as e:
            print(f"  8-K parse failed for {symbol} ({f.accession_no}): {e}")
    return out
```

```python
"""Compute RSI-14 via Wilder smoothing in Pandas and write back into silver.silver_prices_cleaned.rsi_14."""
import pandas as pd

from ingest.common import pg_conn


def wilder_rsi(close: pd.Series, n: int = 14) -> pd.Series:
    delta = close.diff()
    up = delta.clip(lower=0)
    down = -delta.clip(upper=0)
    avg_up = up.ewm(alpha=1 / n, adjust=False).mean()
    avg_down = down.ewm(alpha=1 / n, adjust=False).mean()
    rs = avg_up / avg_down
    return 100 - 100 / (1 + rs)


def main():
    with pg_conn() as conn:
        df = pd.read_sql(
            "SELECT price_key, symbol, trade_date, close_px FROM silver.silver_prices_cleaned ORDER BY symbol, trade_date",
            conn,
        )
        df["rsi_14"] = df.groupby("symbol")["close_px"].transform(lambda s: wilder_rsi(s))
        with conn.cursor() as cur:
            for _, row in df.iterrows():
                if pd.isna(row["rsi_14"]):
                    continue
                cur.execute(
                    "UPDATE silver.silver_prices_cleaned SET rsi_14 = %s WHERE price_key = %s",
                    (float(row["rsi_14"]), row["price_key"]),
                )
        print(f"Updated rsi_14 for {df['rsi_14'].notna().sum()} rows")


if __name__ == "__main__":
    main()
```

```sql
        SUM(net_income)       OVER w AS net_income_ttm,
        SUM(ebitda)           OVER w AS ebitda_ttm,
        SUM(free_cash_flow)   OVER w AS free_cash_flow_ttm,
        AVG(total_assets)     OVER w AS avg_assets_ttm,
        AVG(total_equity)     OVER w AS avg_equity_ttm,
        total_assets, total_liabilities, total_equity, total_debt
    FROM joined
    WINDOW w AS (PARTITION BY symbol ORDER BY filing_date ROWS BETWEEN 3 PRECEDING AND CURRENT ROW)
)
SELECT
    MD5(symbol || '|' || filing_date::text) AS fund_key,
    MD5(LOWER(TRIM(symbol)))                AS company_key,
    symbol,
    filing_date,
    revenue_ttm, gross_profit_ttm, operating_income_ttm, net_income_ttm, ebitda_ttm, free_cash_flow_ttm,
    avg_assets_ttm, avg_equity_ttm,
    total_assets, total_liabilities, total_equity, total_debt,
    CASE WHEN avg_equity_ttm > 0 THEN net_income_ttm / avg_equity_ttm END AS roe,
    CASE WHEN avg_assets_ttm > 0 THEN net_income_ttm / avg_assets_ttm END AS roa,
    CASE WHEN total_equity > 0 THEN total_debt / total_equity END AS debt_eq,
    CASE WHEN revenue_ttm > 0 THEN gross_profit_ttm / revenue_ttm END AS gross_margin,
    CASE WHEN revenue_ttm > 0 THEN operating_income_ttm / revenue_ttm END AS op_margin,
    CASE WHEN avg_assets_ttm > 0 THEN revenue_ttm / avg_assets_ttm END AS asset_turnover,
    filing_date AS as_of_date
FROM ttm
WHERE revenue_ttm IS NOT NULL
```

PE, PB, FCF yield, EV/EBITDA all depend on `market_cap` which we add via join in `gold.fct_feature_panel_daily`.

```python
"""Score news + press articles with FinBERT, write back into Postgres."""
from __future__ import annotations

import pandas as pd
import torch
from tqdm import tqdm
from transformers import AutoModelForSequenceClassification, AutoTokenizer

from ingest.common import pg_conn

MODEL_NAME = "yiyanghkust/finbert-tone"


def load_model():
    tok = AutoTokenizer.from_pretrained(MODEL_NAME)
    model = AutoModelForSequenceClassification.from_pretrained(MODEL_NAME)
    model.eval()
    return tok, model


def score_text(text: str, tok, model) -> float:
    """Return positive_prob - negative_prob as a scalar score in [-1, +1]."""
    if not text or not text.strip():
        return 0.0
    enc = tok(text[:512], return_tensors="pt", truncation=True, max_length=512)
    with torch.no_grad():
        out = model(**enc).logits.softmax(dim=-1).squeeze().tolist()
    # FinBERT tone label order: ['positive', 'negative', 'neutral']
    return float(out[0] - out[1])


def score_table(source_view: str, target_table: str, body_col: str, id_col: str) -> None:
    tok, model = load_model()
    with pg_conn() as conn:
        df = pd.read_sql(f"SELECT {id_col}, symbol, published_at, {body_col} AS body FROM {source_view}", conn)
    df["finbert_score"] = [
        score_text(t, tok, model) for t in tqdm(df["body"].tolist(), desc=target_table)
    ]
    df["finbert_score"] = df["finbert_score"].astype(float)

    with pg_conn() as conn, conn.cursor() as cur:
        cur.execute(f"DROP TABLE IF EXISTS {target_table} CASCADE")
        cur.execute(
            f"""
            CREATE TABLE {target_table} (
              {id_col}       text PRIMARY KEY,
              symbol         text NOT NULL,
              company_key    text NOT NULL,
              published_at   text,
              trade_date     date,
              finbert_score  numeric(8,5)
            )
            """
        )
        for _, r in df.iterrows():
            cur.execute(
                f"""
                INSERT INTO {target_table} ({id_col}, symbol, company_key, published_at, trade_date, finbert_score)
                VALUES (%s, %s, %s, %s, %s, %s)
                """,
                (
                    r[id_col],
                    r["symbol"],
                    None,  # filled by SQL UPDATE next
                    r["published_at"],
                    pd.to_datetime(r["published_at"]).date() if pd.notna(r["published_at"]) else None,
                    r["finbert_score"],
                ),
            )
        cur.execute(f"UPDATE {target_table} SET company_key = md5(lower(trim(symbol)))")
    print(f"Wrote {len(df)} rows to {target_table}")


def main() -> None:
    score_table(
        source_view="datos_masked.news_redacted",
        target_table="silver.silver_news_scored",
        body_col="body_masked",
        id_col="article_id",
    )
    score_table(
        source_view="datos_masked.press_redacted",
        target_table="silver.silver_press_scored",
        body_col="body_masked",
        id_col="press_id",
    )


if __name__ == "__main__":
    main()
```

```python
"""Deterministic 500-token chunker with 50-token overlap. Uses tiktoken for accurate token counting."""
from __future__ import annotations

import tiktoken

ENC = tiktoken.get_encoding("cl100k_base")
CHUNK_TOKENS = 500
OVERLAP_TOKENS = 50


def chunk_text(text: str) -> list[str]:
    if not text:
        return []
    tokens = ENC.encode(text)
    chunks = []
    step = CHUNK_TOKENS - OVERLAP_TOKENS
    for i in range(0, len(tokens), step):
        slice_tokens = tokens[i : i + CHUNK_TOKENS]
        if len(slice_tokens) < 50:  # skip tiny tails
            break
        chunks.append(ENC.decode(slice_tokens))
    return chunks
```

```python
"""Chunk + embed every 10-K and 8-K filing, write to silver + rag_chunks_pgvector."""
from __future__ import annotations

import pandas as pd
from tqdm import tqdm

from ingest.common import pg_conn
from silver_text.chunk import chunk_text
from silver_text.embed import embed


def build_for(source_view: str, target_table: str, source_type_code: str) -> None:
    with pg_conn() as conn:
        df = pd.read_sql(
            f"SELECT accession, symbol, filing_date, body_masked AS body FROM {source_view}",
            conn,
        )

    chunks_rows = []
    for _, row in df.iterrows():
        chunks = chunk_text(row["body"])
        for idx, chunk in enumerate(chunks):
            chunks_rows.append(
                dict(
                    chunk_key=f"{source_type_code}-{row['symbol']}-{row['filing_date']}-c{idx:02d}",
                    accession=row["accession"],
                    symbol=row["symbol"],
                    filing_date=row["filing_date"],
                    chunk_index=idx,
                    n_tokens=len(chunk.split()),  # approximate
                    body_chunk=chunk,
                )
            )

    if not chunks_rows:
        print(f"No chunks for {target_table}")
        return

    chunk_df = pd.DataFrame(chunks_rows)
    print(f"Embedding {len(chunk_df)} chunks for {target_table}...")
    embeddings = embed(chunk_df["body_chunk"].tolist())

    with pg_conn() as conn, conn.cursor() as cur:
        cur.execute(f"DROP TABLE IF EXISTS {target_table} CASCADE")
        cur.execute(
            f"""
            CREATE TABLE {target_table} (
              chunk_key      text PRIMARY KEY,
              accession      text,
              symbol         text,
              company_key    text,
              filing_date    date,
              chunk_index    smallint,
              n_tokens       smallint,
              body_chunk     text,
              as_of_date     date NOT NULL
            )
            """
        )
        for _, r in chunk_df.iterrows():
            cur.execute(
                f"""
                INSERT INTO {target_table}
                  (chunk_key, accession, symbol, company_key, filing_date, chunk_index, n_tokens, body_chunk, as_of_date)
                VALUES (%s, %s, %s, md5(lower(trim(%s))), %s, %s, %s, %s, %s)
                """,
                (
                    r["chunk_key"],
                    r["accession"],
                    r["symbol"],
                    r["symbol"],
                    r["filing_date"],
                    int(r["chunk_index"]),
                    int(r["n_tokens"]),
                    r["body_chunk"],
                    r["filing_date"],  # as_of_date
                ),
            )

        # Embeddings -> gold.rag_chunks_pgvector
        cur.execute(
            """
            CREATE TABLE IF NOT EXISTS gold.rag_chunks_pgvector (
              chunk_key   text PRIMARY KEY,
              embedding   vector(384) NOT NULL,
              inserted_at timestamptz NOT NULL DEFAULT now()
            )
            """
        )
        for emb, key in zip(embeddings, chunk_df["chunk_key"]):
            cur.execute(
                "INSERT INTO gold.rag_chunks_pgvector (chunk_key, embedding) VALUES (%s, %s) ON CONFLICT (chunk_key) DO UPDATE SET embedding = EXCLUDED.embedding",
                (key, emb.tolist()),
            )

    print(f"Wrote {len(chunk_df)} chunks + embeddings for {target_table}")


def main() -> None:
    build_for(
        source_view="datos_masked.filings_10k_redacted",
        target_table="silver.silver_filings_10k_chunked",
        source_type_code="10K",
    )
    build_for(
        source_view="datos_masked.filings_8k_redacted",
        target_table="silver.silver_filings_8k_chunked",
        source_type_code="8K",
    )

    # HNSW index on pgvector
    with pg_conn() as conn, conn.cursor() as cur:
        cur.execute("DROP INDEX IF EXISTS gold.rag_chunks_hnsw_idx")
        cur.execute(
            """
            CREATE INDEX rag_chunks_hnsw_idx
              ON gold.rag_chunks_pgvector
              USING hnsw (embedding vector_cosine_ops)
              WITH (m = 16, ef_construction = 64)
            """
        )
    print("Built HNSW index on rag_chunks_pgvector")


if __name__ == "__main__":
    main()
```

```sql
WITH news_per_day AS (
```

```python
    company_means = df.groupby("company_key")["embedding"].apply(lambda s: np.mean(np.vstack(s.values), axis=0))

            (

                ck,
                df[df["company_key"] == ck]["filing_date"].max(),
                float(reduced[i, 0]),
                float(reduced[i, 1]),
                float(reduced[i, 2]),
                float(reduced[i, 3]),
                float(reduced[i, 4]),
            )
        )

    with pg_conn() as conn, conn.cursor() as cur:
        cur.execute("DROP TABLE IF EXISTS gold.fct_embedding_per_company CASCADE")
```

```python
"""Generate (train_start, train_end, test_start, test_end) folds for walk-forward."""
from __future__ import annotations

from datetime import date, timedelta
from dataclasses import dataclass


@dataclass
class Fold:
    fold_id: str
    train_start: date
    train_end: date
    test_start: date
    test_end: date


def folds(
    start: date = date(2021, 1, 1),
    end: date = date(2025, 12, 31),
    train_years: int = 3,
    test_quarters: int = 1,
    gap_days: int = 5,
) -> list[Fold]:
    out = []
    cursor = start + timedelta(days=365 * train_years)
    test_len = timedelta(days=int(365 / 4 * test_quarters))
    while cursor + test_len <= end:
        train_end = cursor - timedelta(days=gap_days)
        train_start = train_end - timedelta(days=365 * train_years)
        test_start = cursor
        test_end = cursor + test_len
        out.append(
            Fold(
                fold_id=f"{test_start.year}Q{((test_start.month - 1) // 3) + 1}",
                train_start=train_start,
                train_end=train_end,
                test_start=test_start,
                test_end=test_end,
            )
        )
        cursor = cursor + test_len
    return out


if __name__ == "__main__":
    for f in folds():
        print(f.fold_id, f.train_start, f.train_end, "->", f.test_start, f.test_end)
```

```python
"""Rung 0: ARIMA per-stock on log_ret_1d, output prob_up + predicted_class."""
from __future__ import annotations

import math

import numpy as np
import pandas as pd
from pmdarima import auto_arima

import mlflow
from sklearn.metrics import roc_auc_score, accuracy_score

from ingest.common import pg_conn
from models.mlflow_helpers import run as mlflow_run
from models.walkforward import folds


def predict_5d_direction(log_returns: pd.Series) -> tuple[float, int]:
    """Fit ARIMA on log_returns, predict next 5 days cumulative return, return (prob_up, predicted_class)."""
    series = log_returns.dropna()
    if len(series) < 60:
        return 0.5, 0
    model = auto_arima(series, suppress_warnings=True, error_action="ignore",
                       max_p=2, max_d=1, max_q=2, seasonal=False)
    forecast = model.predict(n_periods=5)
    cum = float(np.sum(forecast))
    sigma_5d = float(series.rolling(60).std().iloc[-1]) * math.sqrt(5) if len(series) >= 60 else 0.05
    z = cum / max(sigma_5d, 1e-6)
    prob_up = 1.0 / (1.0 + math.exp(-z))
    return prob_up, int(prob_up > 0.5)


def main() -> None:
    with pg_conn() as conn:
        df = pd.read_sql(
            "SELECT trade_date::date, symbol, log_ret_1d, y_5d_up FROM gold.fct_feature_panel_daily WHERE y_5d_up IS NOT NULL ORDER BY symbol, trade_date",
            conn,
        )

    for fold in folds():
        train = df[(df["trade_date"] >= fold.train_start) & (df["trade_date"] <= fold.train_end)]
        test  = df[(df["trade_date"] >= fold.test_start) & (df["trade_date"] <= fold.test_end)]
        preds, labels = [], []
        for symbol, sub in test.groupby("symbol"):
            train_sub = train[train["symbol"] == symbol]["log_ret_1d"]
            for _, row in sub.iterrows():
                p, c = predict_5d_direction(train_sub)
                preds.append(p)
                labels.append(int(row["y_5d_up"]))
        if not labels:
            continue
        with mlflow_run(rung=0, fold_id=fold.fold_id, model_family="arima", n_features=1) as ml:
            try:
                auc = roc_auc_score(labels, preds)
            except Exception:
                auc = 0.5
            acc = accuracy_score(labels, [int(p > 0.5) for p in preds])
            ml.log_metric("test_auc", auc)
            ml.log_metric("accuracy", acc)
            ml.log_metric("n_test_rows", len(labels))
            print(f"ARIMA fold {fold.fold_id}: AUC={auc:.4f} acc={acc:.4f} n={len(labels)}")


if __name__ == "__main__":
    main()
```

```python
        test  = df[(df["trade_date"] >= fold.test_start) & (df["trade_date"] <= fold.test_end)]
        if test.empty:

        X_test, y_test = test[FULL_FEATURES].fillna(0), test["y_5d_up"]

        clf.fit(X_train, y_train, eval_set=[(X_test, y_test)], verbose=False)

        acc = accuracy_score(y_test, (probs > 0.5).astype(int))

            ml.log_metric("accuracy", acc)
```

```python
"""Walk-forward backtest: long top-5 by prob_up per rung, hold 5 days, 5bp cost. Writes gold.fct_backtest_pnl_daily."""
from __future__ import annotations

import pandas as pd

from ingest.common import pg_conn

TX_COST_BPS = 5
TOP_K = 5
HOLD_DAYS = 5


def main() -> None:
    with pg_conn() as conn:
        preds = pd.read_sql(
            """
            SELECT p.date_key, p.company_key, p.model_rung, p.prob_up, d.full_date AS trade_date
            FROM gold.fct_predictions p
            JOIN gold.dim_date d ON d.date_key = p.date_key
            ORDER BY model_rung, trade_date, prob_up DESC
            """,
            conn,
        )
        # Build returns: 5-day forward return per (symbol, trade_date)
        ret = pd.read_sql(
            """
            SELECT
              p.company_key, p.trade_date,
              p.close_px,
              LEAD(p.close_px, 5) OVER (PARTITION BY p.company_key ORDER BY p.trade_date) AS close_t5
            FROM silver.silver_prices_cleaned p
            """,
            conn,
        )
        ret["fwd_ret_5d"] = (ret["close_t5"] - ret["close_px"]) / ret["close_px"]

    pnl_rows = []
    for rung, group in preds.groupby("model_rung"):
        for date, day_group in group.groupby("trade_date"):
            top = day_group.nlargest(TOP_K, "prob_up")
            merged = top.merge(ret, on=["company_key", "trade_date"], how="left")
            if merged["fwd_ret_5d"].isna().all():
                continue
            gross_ret = float(merged["fwd_ret_5d"].mean())
            cost = TX_COST_BPS / 10000  # one-way 5bp = 0.0005
            net_ret = gross_ret - cost
            pnl_rows.append((str(date), int(rung), TOP_K, gross_ret, cost, net_ret))

    df = pd.DataFrame(pnl_rows, columns=["trade_date", "model_rung", "n_long", "gross_ret", "tx_cost", "net_ret"])
    df["cum_net_ret"] = df.groupby("model_rung")["net_ret"].cumsum()

    with pg_conn() as conn, conn.cursor() as cur:
        cur.execute("""
            DROP TABLE IF EXISTS gold.fct_backtest_pnl_daily;
            CREATE TABLE gold.fct_backtest_pnl_daily (
              fact_pnl_key bigserial PRIMARY KEY,
              trade_date   date,
              date_key     text,
              model_rung   smallint,
              n_long       smallint,
              gross_ret    numeric(18,8),
              tx_cost      numeric(18,8),
              net_ret      numeric(18,8),
              cum_net_ret  numeric(18,8)
            )
        """)
        for _, r in df.iterrows():
            cur.execute(
                "INSERT INTO gold.fct_backtest_pnl_daily (trade_date, date_key, model_rung, n_long, gross_ret, tx_cost, net_ret, cum_net_ret) VALUES (%s, md5(%s), %s, %s, %s, %s, %s, %s)",
                (r["trade_date"], r["trade_date"], r["model_rung"], r["n_long"], float(r["gross_ret"]), float(r["tx_cost"]), float(r["net_ret"]), float(r["cum_net_ret"])),
            )
    print(f"Wrote {len(df)} rows to gold.fct_backtest_pnl_daily")


if __name__ == "__main__":
    main()
```

```python
"""Claude Haiku RAG: take question + retrieved chunks + SHAP, produce explanatory paragraph."""
from __future__ import annotations

import os
import time

import anthropic
import pandas as pd

from ingest.common import pg_conn
from rag.retrieve import retrieve

CLIENT = anthropic.Anthropic(api_key=os.getenv("ANTHROPIC_API_KEY"))
MODEL = "claude-haiku-4-5-20251001"

PROMPT = """You are a financial analyst explaining a model prediction. The model predicted {predicted_class} for {symbol} on {date} with probability {prob_up:.2f}.

Below are the {n_chunks} most relevant text chunks from the company's filings, retrieved by semantic similarity. Each chunk has an ID like [10K-AAPL-2024-c07].

{chunks}

User question: {question}

Instructions:
1. Write one paragraph (4 to 6 sentences) explaining the prediction.
2. Reference at least 2 chunk IDs.
3. Plain language. Do not speculate beyond what the chunks support.
4. If the chunks contradict the prediction, say so honestly.
"""


def answer(question: str, symbol: str, as_of_date: str, prob_up: float, predicted_class: str) -> tuple[str, list[str]]:
    chunks = retrieve(question, symbol, as_of_date, top_k=8)
    if chunks.empty:
        return "No relevant text found for this query.", []
    chunk_block = "\n\n".join([f"[{r.chunk_key}]: {r.body_chunk[:800]}" for _, r in chunks.iterrows()])
    prompt = PROMPT.format(
        symbol=symbol, date=as_of_date, prob_up=prob_up, predicted_class=predicted_class,
        n_chunks=len(chunks), chunks=chunk_block, question=question,
    )
    t0 = time.time()
    resp = CLIENT.messages.create(model=MODEL, max_tokens=600, messages=[{"role": "user", "content": prompt}])
    paragraph = resp.content[0].text
    latency_ms = int((time.time() - t0) * 1000)

    # Audit
    with pg_conn() as conn, conn.cursor() as cur:
        cur.execute(
            "CREATE TABLE IF NOT EXISTS gold.fct_rag_queries (query_id bigserial PRIMARY KEY, ts timestamptz DEFAULT now(), question text, top_k_chunk_ids text[], llm_model text, tokens_in int, tokens_out int, response text, latency_ms int)"
        )
        cur.execute(
            "INSERT INTO gold.fct_rag_queries (question, top_k_chunk_ids, llm_model, tokens_in, tokens_out, response, latency_ms) VALUES (%s, %s, %s, %s, %s, %s, %s)",
            (question, list(chunks["chunk_key"]), MODEL, resp.usage.input_tokens, resp.usage.output_tokens, paragraph, latency_ms),
        )

    return paragraph, list(chunks["chunk_key"])


if __name__ == "__main__":
    para, keys = answer("Why is AAPL predicted up?", "AAPL", "2025-12-01", 0.68, "UP")
    print(para)
    print("Citations:", keys)
```

```python
"""Compute MRR@5 and P@5 over rag/eval_set.jsonl."""
import json
from pathlib import Path

from rag.retrieve import retrieve

EVAL_FILE = Path(__file__).parent / "eval_set.jsonl"


def main() -> None:
    queries = [json.loads(line) for line in EVAL_FILE.read_text().splitlines() if line.strip()]
    mrr_total = 0.0
    p5_total = 0.0
    for q in queries:
        chunks = retrieve(q["question"], q["symbol"], q["as_of_date"], top_k=5)
        ranks = [i for i, k in enumerate(chunks["chunk_key"]) if q["expected_substring"] in k]
        if ranks:
            mrr_total += 1.0 / (ranks[0] + 1)
            p5_total  += len(ranks) / 5.0
    n = len(queries)
    mrr5 = mrr_total / n
    p5 = p5_total / n
    print(f"MRR@5 = {mrr5:.3f}  P@5 = {p5:.3f}  over {n} queries")


if __name__ == "__main__":
    main()
```
